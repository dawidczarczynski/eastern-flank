## detail is the only goal – chamfered parts, individual track shoes, bolts, cables, rails, and a
## weathering shader driven by vertex colours (r = chamfer edge, g = openness, b = tint seed).
## Measurements in metres as in vehicle_frame.gd: `x` from the hull front backwards, `h` above
## the ground, `z` across (positive = vehicle's right).

## Mesh builder in vehicle metres: every primitive writes flat or smooth triangles with the
## weathering vertex colour; `xf` places primitives (metric, rigid), `base` maps model space into
## the pivot the mesh is added to.
class Kit:
	extends RefCounted
	var frame: Frame
	var base: Transform3D
	var st := SurfaceTool.new()
	var xf := Transform3D.IDENTITY
	var cavity := 1.0
	var tint := 0.5
	var count := 0

	func _init(f: Frame, b: Transform3D) -> void:
		frame = f
		base = b
		st.begin(Mesh.PRIMITIVE_TRIANGLES)

	func commit() -> ArrayMesh:
		return st.commit()

	func _map(v: Vector3) -> Vector3:
		var w := xf * v
		return base * frame.p(w.x, w.y, w.z)

	func _map_n(n: Vector3) -> Vector3:
		var w := xf.basis * n
		return (base.basis * Vector3(-w.x, w.y, w.z)).normalized()

	func _set_tint(at: Vector3) -> void:
		var w := xf * at
		tint = fposmod(sin(w.dot(Vector3(12.9898, 78.233, 37.719))) * 43758.5453, 1.0)

	func tri(
		a: Vector3, b: Vector3, c: Vector3, na: Vector3, nb: Vector3, nc: Vector3, e: float
	) -> void:
		var ma := _map(a)
		var mb := _map(b)
		var mc := _map(c)
		var cross := (mb - ma).cross(mc - ma)
		if cross.length_squared() < 1e-16:
			return
		var n0 := _map_n(na)
		var n1 := _map_n(nb)
		var n2 := _map_n(nc)
		if cross.dot(n0 + n1 + n2) > 0.0:
			var t := mb
			mb = mc
			mc = t
			var tn := n1
			n1 = n2
			n2 = tn
		var col := Color(e, cavity, tint)
		for pair: Array in [[ma, n0], [mb, n1], [mc, n2]]:
			st.set_color(col)
			st.set_normal(pair[1])
			st.add_vertex(pair[0])
		count += 1

	## Flat convex polygon with outward normal `n`.
	func poly(pts: Array, n: Vector3, e := 0.0) -> void:
		for i in range(1, pts.size() - 1):
			tri(pts[0], pts[i], pts[i + 1], n, n, n, e)

	## Flat convex polygon, normal from its winding, turned away from `inside`.
	func poly_auto(pts: Array, inside: Vector3, e := 0.0) -> void:
		var a: Vector3 = pts[0]
		var n := ((pts[1] as Vector3) - a).cross((pts[2] as Vector3) - a)
		if n.length_squared() < 1e-14 and pts.size() > 3:
			n = ((pts[2] as Vector3) - a).cross((pts[3] as Vector3) - a)
		if n.length_squared() < 1e-14:
			return
		var mid := Vector3.ZERO
		for p: Vector3 in pts:
			mid += p
		mid /= pts.size()
		if n.dot(mid - inside) < 0.0:
			n = -n
		poly(pts, n.normalized(), e)

	## Chamfered box centred at `c` (size `s`, chamfer `b`), rotated by `rot`.
	func cbox(c: Vector3, s: Vector3, b: float, rot := Basis.IDENTITY) -> void:
		var saved := xf
		xf = xf * Transform3D(rot, c)
		_set_tint(Vector3.ZERO)
		var h := s / 2.0
		b = minf(b, minf(h.x, minf(h.y, h.z)) * 0.9)
		var i := h - Vector3.ONE * b
		for a in 3:
			var u := (a + 1) % 3
			var v := (a + 2) % 3
			for sa: float in [-1.0, 1.0]:
				var pts := []
				for corner: Vector2 in [
					Vector2(-1, -1), Vector2(1, -1), Vector2(1, 1), Vector2(-1, 1)
				]:
					var p := Vector3.ZERO
					p[a] = sa * h[a]
					p[u] = corner.x * i[u]
					p[v] = corner.y * i[v]
					pts.append(p)
				var n := Vector3.ZERO
				n[a] = sa
				poly(pts, n)
		if b > 0.0:
			for a in 3:
				var u := (a + 1) % 3
				var v := (a + 2) % 3
				for sa: float in [-1.0, 1.0]:
					for su: float in [-1.0, 1.0]:
						var p1 := Vector3.ZERO
						p1[a] = sa * h[a]
						p1[u] = su * i[u]
						var p2 := Vector3.ZERO
						p2[a] = sa * i[a]
						p2[u] = su * h[u]
						var q0 := p1
						var q1 := p2
						var q2 := p2
						var q3 := p1
						q0[v] = -i[v]
						q1[v] = -i[v]
						q2[v] = i[v]
						q3[v] = i[v]
						var q := [q0, q1, q2, q3]
						var n := Vector3.ZERO
						n[a] = sa
						n[u] = su
						poly(q, n.normalized(), 1.0)
			for sx: float in [-1.0, 1.0]:
				for sy: float in [-1.0, 1.0]:
					for sz: float in [-1.0, 1.0]:
						var pts := [
							Vector3(sx * h.x, sy * i.y, sz * i.z),
							Vector3(sx * i.x, sy * h.y, sz * i.z),
							Vector3(sx * i.x, sy * i.y, sz * h.z),
						]
						poly(pts, Vector3(sx, sy, sz).normalized(), 1.0)
		xf = saved

	## Soft superellipsoid "pillow" (bags, tarps): box-like with rounded, smooth-shaded sides.
	func pillow(c: Vector3, s: Vector3, rot := Basis.IDENTITY, e := 0.35, rings := 10, sides := 16) -> void:
		var saved := xf
		xf = xf * Transform3D(rot, c)
		_set_tint(Vector3.ZERO)
		var h := s / 2.0
		var grid := []
		var normals := []
		for i in rings + 1:
			var lat := lerpf(-PI / 2.0, PI / 2.0, float(i) / rings)
			var row := []
			var nrow := []
			for j in sides:
				var lon := TAU * j / sides
				var d := Vector3(cos(lat) * cos(lon), sin(lat), cos(lat) * sin(lon))
				var q := Vector3(signf(d.x) * pow(absf(d.x), e), signf(d.y) * pow(absf(d.y), e), signf(d.z) * pow(absf(d.z), e))
				var p := q * h
				var n := Vector3(signf(d.x) * pow(absf(d.x), 2.0 - e) / h.x, signf(d.y) * pow(absf(d.y), 2.0 - e) / h.y, signf(d.z) * pow(absf(d.z), 2.0 - e) / h.z)
				row.append(p)
				nrow.append(n.normalized() if n.length_squared() > 1e-12 else d)
			grid.append(row)
			normals.append(nrow)
		for i in rings:
			for j in sides:
				var k := (j + 1) % sides
				tri(grid[i][j], grid[i][k], grid[i + 1][k], normals[i][j], normals[i][k], normals[i + 1][k], 0.0)
				tri(grid[i][j], grid[i + 1][k], grid[i + 1][j], normals[i][j], normals[i + 1][k], normals[i + 1][j], 0.0)
		xf = saved

	## Box spanning x0..x1, h0..h1, z0 +- hw (vehicle axes), chamfer `b`.
	func slab(x0: float, x1: float, h0: float, h1: float, hw: float, z0 := 0.0, b := 0.02) -> void:
		cbox(
			Vector3((x0 + x1) / 2.0, (h0 + h1) / 2.0, z0),
			Vector3(absf(x1 - x0), h1 - h0, hw * 2.0),
			b
		)

	func pair(x0: float, x1: float, h0: float, h1: float, hw: float, z: float, b := 0.02) -> void:
		for s: float in [-1.0, 1.0]:
			slab(x0, x1, h0, h1, hw, s * z, b)

	## Lathe around local +Y of `rot` at `c`: `prof` = Vector2(radius, y) from the top down around
	## the outside. Short diagonal segments count as chamfers (edge wear).
	func lathe(c: Vector3, rot: Basis, prof: Array, sides: int, smooth := true) -> void:
		var saved := xf
		xf = xf * Transform3D(rot, c)
		_set_tint(Vector3.ZERO)
		for j in prof.size() - 1:
			var p0: Vector2 = prof[j]
			var p1: Vector2 = prof[j + 1]
			var d := p1 - p0
			if d.length() < 1e-6:
				continue
			var n2 := Vector2(-d.y, d.x).normalized()
			var e := 1.0 if absf(d.x) > 1e-5 and absf(d.y) > 1e-5 and d.length() < 0.06 else 0.0
			for k in sides:
				var t0 := TAU * k / sides
				var t1 := TAU * (k + 1) / sides
				var c0 := Vector3(cos(t0), 0.0, sin(t0))
				var c1 := Vector3(cos(t1), 0.0, sin(t1))
				var v00 := c0 * p0.x + Vector3(0, p0.y, 0)
				var v01 := c1 * p0.x + Vector3(0, p0.y, 0)
				var v10 := c0 * p1.x + Vector3(0, p1.y, 0)
				var v11 := c1 * p1.x + Vector3(0, p1.y, 0)
				var n0 := c0 * n2.x + Vector3(0, n2.y, 0)
				var n1 := c1 * n2.x + Vector3(0, n2.y, 0)
				if not smooth:
					var cm := (c0 + c1).normalized()
					n0 = cm * n2.x + Vector3(0, n2.y, 0)
					n1 = n0
				tri(v00, v01, v11, n0, n1, n1, e)
				tri(v00, v11, v10, n0, n1, n0, e)
		xf = saved

	## Cylinder of radius `r`, `length` along `dir` (vehicle axes), centred at `c`, chamfer `b`.
	func rod(c: Vector3, dir: Vector3, r: float, length: float, sides := 12, b := 0.0) -> void:
		var hl := length / 2.0
		var prof: Array
		if b > 0.0:
			prof = [
				Vector2(0, hl),
				Vector2(r - b, hl),
				Vector2(r, hl - b),
				Vector2(r, -hl + b),
				Vector2(r - b, -hl),
				Vector2(0, -hl)
			]
		else:
			prof = [Vector2(0, hl), Vector2(r, hl), Vector2(r, -hl), Vector2(0, -hl)]
		lathe(c, Frame.axis_basis(dir), prof, sides)

	## Hex bolt head on a surface at `c` facing `dir`.
	func bolt(c: Vector3, dir: Vector3, r := 0.022) -> void:
		var hgt := r * 0.7
		lathe(
			c,
			Frame.axis_basis(dir),
			[
				Vector2(0, hgt),
				Vector2(r, hgt),
				Vector2(r, 0.0)
			],
			6,
			false
		)

	## Rings of points connected in order (same count, convex cross-sections), capped at both ends;
	## `edges[i]` flags band i as a chamfer.
	func stack(rings: Array, edges: Array) -> void:
		var centre := Vector3.ZERO
		var total := 0
		for ring: Array in rings:
			for p: Vector3 in ring:
				centre += p
				total += 1
		centre /= total
		_set_tint(centre)
		var n := (rings[0] as Array).size()
		for r in rings.size() - 1:
			var a: Array = rings[r]
			var b: Array = rings[r + 1]
			for i in n:
				var j := (i + 1) % n
				poly_auto([a[i], a[j], b[j], b[i]], centre, edges[r])
		poly_auto(rings[0], centre)
		poly_auto(rings[rings.size() - 1], centre)

	## Loft of plan outlines (Vector2(x, z)) from h0 to h1, the roof edge chamfered by `b`.
	func loft(bottom: Array, top: Array, h0: float, h1: float, b: float) -> void:
		var inner := _offset(_ccw(top), -b)
		var lo := []
		var mid := []
		var hi := []
		for p: Vector2 in _ccw(bottom):
			lo.append(Vector3(p.x, h0, p.y))
		for p: Vector2 in _ccw(top):
			mid.append(Vector3(p.x, h1 - b, p.y))
		for p: Vector2 in inner:
			hi.append(Vector3(p.x, h1, p.y))
		stack([lo, mid, hi], [0.0, 1.0])

	## Prism of a polygon in the (U, V) plane from w0 to w1 along W, both rims chamfered by `b`;
	## works for non-convex outlines (gears, skirts).
	func plate(
		outline: Array,
		o: Vector3,
		u: Vector3,
		v: Vector3,
		w: Vector3,
		w0: float,
		w1: float,
		b := 0.0
	) -> void:
		var ring := _ccw(outline)
		var inner := _offset(ring, -b) if b > 0.0 else ring
		var at := func(p: Vector2, depth: float) -> Vector3:
			return o + u * p.x + v * p.y + w * depth
		_set_tint(o + u * ring[0].x + v * ring[0].y)
		var n := ring.size()
		for i in n:
			var j := (i + 1) % n
			var d: Vector2 = (ring[j] as Vector2) - (ring[i] as Vector2)
			var n2 := Vector2(d.y, -d.x).normalized()
			var side := (u * n2.x + v * n2.y).normalized()
			poly(
				[
					at.call(ring[i], w0 + b),
					at.call(ring[j], w0 + b),
					at.call(ring[j], w1 - b),
					at.call(ring[i], w1 - b)
				],
				side
			)
			if b > 0.0:
				poly(
					[
						at.call(ring[i], w1 - b),
						at.call(ring[j], w1 - b),
						at.call(inner[j], w1),
						at.call(inner[i], w1)
					],
					(side + w).normalized(),
					1.0
				)
				poly(
					[
						at.call(ring[i], w0 + b),
						at.call(ring[j], w0 + b),
						at.call(inner[j], w0),
						at.call(inner[i], w0)
					],
					(side - w).normalized(),
					1.0
				)
		var idx := Geometry2D.triangulate_polygon(PackedVector2Array(inner))
		for t in range(0, idx.size(), 3):
			var a: Vector2 = inner[idx[t]]
			var bb: Vector2 = inner[idx[t + 1]]
			var c: Vector2 = inner[idx[t + 2]]
			poly([at.call(a, w1), at.call(bb, w1), at.call(c, w1)], w)
			poly([at.call(a, w0), at.call(bb, w0), at.call(c, w0)], -w)

	## Side profile (Vector2(x, h)) extruded across z0 +- hw.
	func profile(points: Array, hw: float, z0 := 0.0, b := 0.03) -> void:
		plate(points, Vector3(0, 0, z0), Vector3.RIGHT, Vector3.UP, Vector3.BACK, -hw, hw, b)

	## Round tube along a polyline (cables, rails, handles), smooth, capped.
	func tube(path: Array, r: float, sides := 8) -> void:
		var n := path.size()
		var rings := []
		var normals := []
		var side := Vector3.ZERO
		_set_tint(path[0])
		for i in n:
			var t: Vector3
			if i == 0:
				t = (path[1] as Vector3) - (path[0] as Vector3)
			elif i == n - 1:
				t = (path[i] as Vector3) - (path[i - 1] as Vector3)
			else:
				t = (path[i + 1] as Vector3) - (path[i - 1] as Vector3)
			t = t.normalized()
			if i == 0:
				var ref := Vector3.UP if absf(t.dot(Vector3.UP)) < 0.9 else Vector3.RIGHT
				side = t.cross(ref).normalized()
			else:
				side = (side - t * side.dot(t)).normalized()
			var up := side.cross(t)
			var ring := []
			var nr := []
			for k in sides:
				var a := TAU * k / sides
				var dir := side * cos(a) + up * sin(a)
				ring.append((path[i] as Vector3) + dir * r)
				nr.append(dir)
			rings.append(ring)
			normals.append(nr)
		for i in n - 1:
			for k in sides:
				var l := (k + 1) % sides
				tri(
					rings[i][k],
					rings[i][l],
					rings[i + 1][l],
					normals[i][k],
					normals[i][l],
					normals[i + 1][l],
					0.0
				)
				tri(
					rings[i][k],
					rings[i + 1][l],
					rings[i + 1][k],
					normals[i][k],
					normals[i + 1][l],
					normals[i + 1][k],
					0.0
				)
		var t0 := ((path[0] as Vector3) - (path[1] as Vector3)).normalized()
		var t1 := ((path[n - 1] as Vector3) - (path[n - 2] as Vector3)).normalized()
		for k in range(1, sides - 1):
			tri(rings[0][0], rings[0][k], rings[0][k + 1], t0, t0, t0, 1.0)
			tri(rings[n - 1][0], rings[n - 1][k], rings[n - 1][k + 1], t1, t1, t1, 1.0)

	## Arc of `segments` points: centre `c`, radius `r`, from angle a0 to a1 in the plane (e1, e2).
	static func arc(
		c: Vector3, r: float, e1: Vector3, e2: Vector3, a0: float, a1: float, segments: int
	) -> Array:
		var pts := []
		for i in segments + 1:
			var a := lerpf(a0, a1, float(i) / segments)
			pts.append(c + (e1 * cos(a) + e2 * sin(a)) * r)
		return pts

	static func _ccw(outline: Array) -> Array:
		var area := 0.0
		for i in outline.size():
			area += (outline[i] as Vector2).cross(outline[(i + 1) % outline.size()])
		if area >= 0.0:
			return outline.duplicate()
		var reversed := outline.duplicate()
		reversed.reverse()
		return reversed

	static func _offset(path: Array, distance: float) -> Array:
		var result := []
		var n := path.size()
		for i in n:
			var before: Vector2 = (path[i] as Vector2) - (path[(i - 1 + n) % n] as Vector2)
			var after: Vector2 = (path[(i + 1) % n] as Vector2) - (path[i] as Vector2)
			var na := Vector2(before.y, -before.x).normalized()
			var nb := Vector2(after.y, -after.x).normalized()
			var bis := (na + nb).normalized()
			var scale := 1.0 / maxf(bis.dot(na), 0.35)
			result.append((path[i] as Vector2) + bis * distance * scale)
		return result


## One kit per material for one pivot.
class Kits:
	extends RefCounted
	var frame: Frame
	var base: Transform3D
	var kits := {}

	func _init(f: Frame, b: Transform3D) -> void:
		frame = f
		base = b

	func k(material: StringName) -> Kit:
		if not kits.has(material):
			kits[material] = Kit.new(frame, base)
		return kits[material]

	func add_to(parent: Node3D, materials: Dictionary) -> void:
		for name: StringName in kits:
			var kit: Kit = kits[name]
			if kit.count > 0:
				ModelParts.add(parent, kit.commit(), materials[name], Vector3.ZERO)


static func build(faction: FactionDefinition) -> Node3D:
	var g := Node3D.new()
	var frame := Frame.new(K, (HULL_LENGTH - GUN_OVERHANG) / 2.0)
	var m := _materials(faction)
	var hull := Kits.new(frame, Transform3D.IDENTITY)
	_hull(hull)
	_engine_deck(hull)
	_rear(hull)
	_front(hull)
	_running_gear(hull)
	_track(hull)
	_skirts(hull)
	hull.add_to(g, m)
	var turret := frame.add_turret(g, TURRET_X, DECK_H)
	var tk := Kits.new(frame, Frame.to_local(turret))
	_turret(tk)
	_turret_roof(tk)
	_crows(tk)
	_bustle(tk)
	_turret_sides(tk)
	tk.add_to(turret, m)
	var gun := frame.add_gun(turret, TRUNNION, -GUN_OVERHANG - 0.05)
	var gk := Kits.new(frame, Frame.to_local(gun))
	_gun(gk)
	gk.add_to(gun, m)
	return g


static func _paint(colors: Array, mode: int, params: Dictionary) -> ShaderMaterial:
	var mat := ShaderMaterial.new()
	mat.shader = PAINT
	mat.set_shader_parameter(&"color_a", colors[0])
	mat.set_shader_parameter(&"color_b", colors[1] if colors.size() > 1 else colors[0])
	mat.set_shader_parameter(&"color_c", colors[2] if colors.size() > 2 else colors[0])
	mat.set_shader_parameter(&"pattern_mode", mode)
	for key: String in params:
		mat.set_shader_parameter(StringName(key), params[key])
	return mat


static func _materials(faction: FactionDefinition) -> Dictionary:
	var camo := ModelParts.camo_colors(faction)
	var glass := ModelParts.std_material(Color("#0d1a1f"), 0.05, 0.6)
	var lens := ModelParts.std_material(Color("#c9c4b0"), 0.1, 0.3)
	var red_lens := ModelParts.std_material(Color("#6a1410"), 0.15, 0.2)
	return {
		&"body": _paint(camo, 0, {}),
		&"dark":
		_paint(

static func _wheel_basis(s: float) -> Basis:
	return Basis(Vector3.RIGHT, PI / 2.0 if s > 0.0 else -PI / 2.0)


static func _running_gear(h: Kits) -> void:
	var body := h.k(&"body")
	var rubber := h.k(&"rubber")
	var dark := h.k(&"dark")
	var steel := h.k(&"steel")
	for s: float in [-1.0, 1.0]:
		var rot := _wheel_basis(s)
		for x: float in ROAD_X:
			# Double road wheel: the outer half detailed (dished disc, tyre, hub cap, bolts); the
			# inner half is hidden behind it and the track, so a plain drum.
			dark.rod(Vector3(x, ROAD_H, s * (TRACK_Z - 0.08)), Vector3.BACK, 0.3, 0.1, 8)
			for half: float in [1.0]:
				var zc := s * (TRACK_Z + half * 0.08)
				var r := rot if half > 0.0 else _wheel_basis(-s)
				var c := Vector3(x, ROAD_H, zc)
				body.lathe(
					c,
					r,
					[
						Vector2(0.0, 0.17),
						Vector2(0.08, 0.17),
						Vector2(0.1, 0.13),
						Vector2(0.235, 0.09),
						Vector2(0.27, 0.11),
						Vector2(0.27, 0.0)
					],
					12
				)
				rubber.lathe(
					c,
					r,
					[
						Vector2(0.27, 0.105),
						Vector2(0.3, 0.105),
						Vector2(0.32, 0.085),
						Vector2(0.32, 0.02),
						Vector2(0.3, 0.0)
					],
					12
				)
				if half > 0.0:
					for k in 6:
						var a := TAU * k / 6.0

							c + Vector3(cos(a) * 0.125, sin(a) * 0.125, s * 0.12),
							Vector3(0, 0, s),

						)

						r,

							Vector2(0.0, 0.215),
							Vector2(0.045, 0.215),
							Vector2(0.055, 0.2),
							Vector2(0.055, 0.16),
							Vector2(0.0, 0.16)
						],

					)
			# Road arm trailing from its hull pivot (only its lower end shows between the wheels).
			var pivot := Vector3(x - 0.42, 0.62, s * 1.09)
			var hub := Vector3(x, ROAD_H, s * 1.09)
			var d := hub - pivot
			var arm_basis := Basis(Vector3.BACK, atan2(d.y, d.x))
			dark.cbox((pivot + hub) / 2.0, Vector3(d.length() + 0.12, 0.12, 0.09), 0.0, arm_basis)
		# Return rollers: behind the skirts, only there to carry the top run.

		body.lathe(
			Vector3(IDLER.x, IDLER.y, s * (TRACK_Z + 0.08)),
			rot,
			[
				Vector2(0.0, 0.15),
				Vector2(0.07, 0.15),

			],

		)
		dark.cbox(
			Vector3(IDLER.x + 0.25, 0.85, s * 1.1),
			Vector3(0.5, 0.14, 0.1),
			0.0,
			Basis(Vector3.BACK, -0.35)
		)
		# Drive sprocket: two toothed rings, hub with bolts and cover.
		for half: float in [-1.0, 1.0]:
			var w := s * (TRACK_Z + half * 0.1)
			var gear := []
			for p in ModelParts.gear_outline(0.3, SPROCKET.z, 11):
				gear.append(p)
			steel.plate(
				gear,
				Vector3(SPROCKET.x, SPROCKET.y, w),
				Vector3.RIGHT,
				Vector3.UP,
				Vector3.BACK,
				-0.025,
				0.025,
				0.0
			)
			body.lathe(
				Vector3(SPROCKET.x, SPROCKET.y, w),
				rot,
				[
					Vector2(0.0, 0.035),
					Vector2(0.28, 0.035),
					Vector2(0.3, 0.02),
					Vector2(0.3, -0.02),
					Vector2(0.28, -0.035),
					Vector2(0.0, -0.035)
				],
				16
			)
		var hub_c := Vector3(SPROCKET.x, SPROCKET.y, s * (TRACK_Z + 0.1))
		body.lathe(
			hub_c,
			rot,
			[
				Vector2(0.0, 0.16),

				Vector3(-cos(a) * 0.3, -sin(a) * 0.3, s).normalized(),
				0.016
			)
		body.rod(Vector3(SPROCKET.x, SPROCKET.y, s * 1.2), Vector3.BACK, 0.24, 0.12, 12)


## Track: individual shoes (steel body, twin rubber pads, end connectors with wedges, centre guide
## horns) stepped along the loop around the wheels.
static func _track(h: Kits) -> void:
	var circles: Array[Vector3] = []
	for x: float in ROAD_X:
		circles.append(Vector3(x, ROAD_H, ROAD_R - 0.0))
	for x: float in ROLLERS:
		circles.append(Vector3(x, ROLLER_H, 0.1))
	circles.append(Vector3(SPROCKET.x, SPROCKET.y, 0.3))
	circles.append(Vector3(IDLER.x, IDLER.y, IDLER.z))
	var loop := ModelParts.circle_hull(circles, 24)
	var track := h.k(&"track")
	var rubber := h.k(&"rubber")
	var steel := h.k(&"steel")
	for s: float in [-1.0, 1.0]:
		for station: Array in ModelParts.path_stations(loop, 0.193):
			var p: Vector2 = station[0]
			var t: Vector2 = station[1]
			var out := Vector2(t.y, -t.x)
			var basis := Basis(
				Vector3(t.x, t.y, 0.0), Vector3(out.x, out.y, 0.0), Vector3(0, 0, -1)
			)
			var c := Vector3(p.x, p.y, s * TRACK_Z)
			var o3 := Vector3(out.x, out.y, 0.0)
			# Top run behind the skirts is never seen: no shoes there.
			if p.y > 0.9 and p.x > SKIRT_X0 + 0.5 and p.x < SKIRT_X1:
				continue
			# Lite shoe: plain steel body + rubber pad (the guide horns and end connectors only show
			# from below).
			track.cbox(c + o3 * 0.035, Vector3(0.186, 0.05, TRACK_W - 0.02), 0.0, basis)

static func _skirts(h: Kits) -> void:
	var body := h.k(&"body")
	var rubber := h.k(&"rubber")
	var steel := h.k(&"steel")
	var panels := 7
	var step := (SKIRT_X1 - SKIRT_X0) / panels
	for s: float in [-1.0, 1.0]:
		for i in panels:
			var x0 := SKIRT_X0 + i * step + 0.006
			var x1 := SKIRT_X0 + (i + 1) * step - 0.006
			var thick := 0.13 if i < 2 else 0.05
			var outline := [
				Vector2(x0, SKIRT_H1),
				Vector2(x1, SKIRT_H1),
				Vector2(x1, SKIRT_H0),
				Vector2(x0, SKIRT_H0)
			]
			if i == 0:
				outline = [
					Vector2(x0, SKIRT_H1),
					Vector2(x1, SKIRT_H1),
					Vector2(x1, SKIRT_H0),
					Vector2(x0 + 0.4, SKIRT_H0),
					Vector2(x0, SKIRT_H0 + 0.3)
				]
			var w0 := SKIRT_Z - thick
			body.plate(
				outline,
				Vector3(0, 0, 0),
				Vector3.RIGHT,
				Vector3.UP,
				Vector3.BACK * s,
				w0,
				SKIRT_Z,
				0.018
			)
			var face := Vector3(0, 0, s)
			if i < 2:
				# Armoured panels: bolt grid.
				for row in 3:
					for col in 4:
						var bx := x0 + 0.12 + col * (x1 - x0 - 0.24) / 3.0
						var bh := SKIRT_H1 - 0.1 - row * 0.2
						if i == 0 and bh < SKIRT_H0 + 0.3 * (1.0 - (bx - x0) / 0.4) + 0.08:
							continue
						body.bolt(Vector3(bx, bh, s * SKIRT_Z), face, 0.03)
			else:
				# Thin panels: rubber edge strip, top bolt row, side stiffener.
				rubber.slab(
					x0 + 0.01,
					x1 - 0.01,
					SKIRT_H0 - 0.04,
					SKIRT_H0 + 0.02,
					0.02,
					s * (SKIRT_Z - 0.03),
					0.006
				)
				for col in 4:
					body.bolt(
						Vector3(
							x0 + 0.1 + col * (x1 - x0 - 0.2) / 3.0, SKIRT_H1 - 0.06, s * SKIRT_Z
						),
						face,
						0.02
					)
				body.slab(
					x0 + 0.04,
					x1 - 0.04,
					SKIRT_H0 + 0.25,
					SKIRT_H0 + 0.28,
					0.012,
					s * (SKIRT_Z + 0.008),
					0.004
				)
			# Hinge knuckles under the fender.
			for hx: float in [x0 + 0.15, x1 - 0.15]:
				steel.rod(
					Vector3(hx, SKIRT_H1 - 0.02, s * (SKIRT_Z - thick / 2.0)),
					Vector3.RIGHT,
					0.025,
					0.12,
					8
				)
		# Skirt top rail under the fender lip.
		body.slab(SKIRT_X0, SKIRT_X1, SKIRT_H1, SKIRT_H1 + 0.04, 0.06, s * (SKIRT_Z - 0.06), 0.01)
