## EXPERIMENT (not a game model): how far the procedural approach goes for an infantry squad –
## smooth swept bodies (elliptic cross-sections), soft gear (superellipsoids), chamfered weapons
## and the showcase paint shader. Same squad layout, poses and kits as pl_mech_infantry.
## Metres: x forward, y up, z right, feet on y = 0; 1 m = 0.1 model units.

const LABEL_HEIGHT := 0.5
const PAINT := preload("res://assets/shaders/showcase_paint.gdshader")
const M := 0.1
const HIP := 0.92
const HIP_KNEEL := 0.55
const SOLDIER_SCALE := 1.75
const RIFLE_BUTT := Vector3(0.12, 1.34, 0.11)
const RIFLE := 0
const RPG := 1
const MG := 2
const SQUAD: Array = [
	[0.12, 0.0, false, RIFLE],
	[-0.03, 0.15, true, RPG],
	[-0.04, -0.15, false, MG],
	[-0.19, 0.06, false, RIFLE],
	[-0.16, -0.13, true, RIFLE],
]


## Mesh builder in soldier metres with the weathering vertex colour (r = edge, g = openness,
## b = tint seed).
class Kit:
	extends RefCounted
	var st := SurfaceTool.new()
	var xf := Transform3D.IDENTITY
	var cavity := 1.0
	var tint := 0.5
	var count := 0

	func _init() -> void:
		st.begin(Mesh.PRIMITIVE_TRIANGLES)

	func commit() -> ArrayMesh:
		return st.commit()

	func _set_tint(at: Vector3) -> void:
		var w := xf * at
		tint = fposmod(sin(w.dot(Vector3(12.9898, 78.233, 37.719))) * 43758.5453, 1.0)

	func tri(
		a: Vector3, b: Vector3, c: Vector3, na: Vector3, nb: Vector3, nc: Vector3, e: float
	) -> void:
		var ma := (xf * a) * M
		var mb := (xf * b) * M
		var mc := (xf * c) * M
		var cross := (mb - ma).cross(mc - ma)
		if cross.length_squared() < 1e-18:
			return
		var n0 := (xf.basis * na).normalized()
		var n1 := (xf.basis * nb).normalized()
		var n2 := (xf.basis * nc).normalized()
		if cross.dot(n0 + n1 + n2) > 0.0:
			var t := mb
			mb = mc
			mc = t
			var tn := n1
			n1 = n2
			n2 = tn
		var col := Color(e, cavity, tint)
		for pair: Array in [[ma, n0], [mb, n1], [mc, n2]]:
			st.set_color(col)
			st.set_normal(pair[1])
			st.add_vertex(pair[0])
		count += 1

	func poly(pts: Array, n: Vector3, e := 0.0) -> void:
		for i in range(1, pts.size() - 1):
			tri(pts[0], pts[i], pts[i + 1], n, n, n, e)

	## Chamfered box centred at `c`, size `s`, chamfer `b`, rotated by `rot`.
	func cbox(c: Vector3, s: Vector3, b: float, rot := Basis.IDENTITY) -> void:
		var saved := xf
		xf = xf * Transform3D(rot, c)
		_set_tint(Vector3.ZERO)
		var h := s / 2.0
		b = minf(b, minf(h.x, minf(h.y, h.z)) * 0.9)
		var i := h - Vector3.ONE * b
		for a in 3:
			var u := (a + 1) % 3
			var v := (a + 2) % 3
			for sa: float in [-1.0, 1.0]:
				var pts := []
				for corner: Vector2 in [
					Vector2(-1, -1), Vector2(1, -1), Vector2(1, 1), Vector2(-1, 1)
				]:
					var p := Vector3.ZERO
					p[a] = sa * h[a]
					p[u] = corner.x * i[u]
					p[v] = corner.y * i[v]
					pts.append(p)
				var n := Vector3.ZERO
				n[a] = sa
				poly(pts, n)
		if b > 0.0:
			for a in 3:
				var u := (a + 1) % 3
				var v := (a + 2) % 3
				for sa: float in [-1.0, 1.0]:
					for su: float in [-1.0, 1.0]:
						var p1 := Vector3.ZERO
						p1[a] = sa * h[a]
						p1[u] = su * i[u]
						var p2 := Vector3.ZERO
						p2[a] = sa * i[a]
						p2[u] = su * h[u]
						var q0 := p1
						var q1 := p2
						var q2 := p2
						var q3 := p1
						q0[v] = -i[v]
						q1[v] = -i[v]
						q2[v] = i[v]
						q3[v] = i[v]
						var n := Vector3.ZERO
						n[a] = sa
						n[u] = su
						poly([q0, q1, q2, q3], n.normalized(), 1.0)
			for sx: float in [-1.0, 1.0]:
				for sy: float in [-1.0, 1.0]:
					for sz: float in [-1.0, 1.0]:
						var pts := [
							Vector3(sx * h.x, sy * i.y, sz * i.z),
							Vector3(sx * i.x, sy * h.y, sz * i.z),
							Vector3(sx * i.x, sy * i.y, sz * h.z),
						]
						poly(pts, Vector3(sx, sy, sz).normalized(), 1.0)
		xf = saved

	## Lathe around local +Y of `rot` at `c`: `prof` = Vector2(radius, y) top down.
	func lathe(c: Vector3, rot: Basis, prof: Array, sides: int) -> void:
		var saved := xf
		xf = xf * Transform3D(rot, c)
		_set_tint(Vector3.ZERO)
		for j in prof.size() - 1:
			var p0: Vector2 = prof[j]
			var p1: Vector2 = prof[j + 1]
			var d := p1 - p0
			if d.length() < 1e-6:
				continue
			var n2 := Vector2(-d.y, d.x).normalized()
			var e := 1.0 if absf(d.x) > 1e-5 and absf(d.y) > 1e-5 and d.length() < 0.02 else 0.0
			for k in sides:
				var t0 := TAU * k / sides
				var t1 := TAU * (k + 1) / sides
				var c0 := Vector3(cos(t0), 0.0, sin(t0))
				var c1 := Vector3(cos(t1), 0.0, sin(t1))
				var n0 := c0 * n2.x + Vector3(0, n2.y, 0)
				var n1 := c1 * n2.x + Vector3(0, n2.y, 0)
				var v00 := c0 * p0.x + Vector3(0, p0.y, 0)
				var v01 := c1 * p0.x + Vector3(0, p0.y, 0)
				var v10 := c0 * p1.x + Vector3(0, p1.y, 0)
				var v11 := c1 * p1.x + Vector3(0, p1.y, 0)
				tri(v00, v01, v11, n0, n1, n1, e)
				tri(v00, v11, v10, n0, n1, n0, e)
		xf = saved

	## Superellipsoid centred at `c` (size `s`, rotated by `rot`); e = 1 is an ellipsoid, smaller
	## is boxier with rounded edges.
	func pillow(
		c: Vector3, s: Vector3, rot := Basis.IDENTITY, e := 0.4, rings := 8, sides := 12
	) -> void:
		var saved := xf
		xf = xf * Transform3D(rot, c)
		_set_tint(Vector3.ZERO)
		var h := s / 2.0
		var grid := []
		var normals := []
		for i in rings + 1:
			var lat := lerpf(-PI / 2.0, PI / 2.0, float(i) / rings)
			var row := []
			var nrow := []
			for j in sides:
				var lon := TAU * j / sides
				var d := Vector3(cos(lat) * cos(lon), sin(lat), cos(lat) * sin(lon))
				var q := Vector3(
					signf(d.x) * pow(absf(d.x), e),
					signf(d.y) * pow(absf(d.y), e),
					signf(d.z) * pow(absf(d.z), e)
				)
				var n := Vector3(
					signf(d.x) * pow(absf(d.x), 2.0 - e) / h.x,
					signf(d.y) * pow(absf(d.y), 2.0 - e) / h.y,
					signf(d.z) * pow(absf(d.z), 2.0 - e) / h.z
				)
				row.append(q * h)
				nrow.append(n.normalized() if n.length_squared() > 1e-12 else d)
			grid.append(row)
			normals.append(nrow)
		for i in rings:
			for j in sides:
				var k := (j + 1) % sides
				tri(
					grid[i][j],
					grid[i][k],
					grid[i + 1][k],
					normals[i][j],
					normals[i][k],
					normals[i + 1][k],
					0.0
				)
				tri(
					grid[i][j],
					grid[i + 1][k],
					grid[i + 1][j],
					normals[i][j],
					normals[i + 1][k],
					normals[i + 1][j],
					0.0
				)
		xf = saved

	## Elliptic rings along `path`: `radii[i]` = Vector2(r along the side axis, r along the other
	## one); the side axis starts as `side0` and is carried along the path. Smooth, capped.
	func sweep(path: Array, radii: Array, sides := 12, side0 := Vector3.BACK) -> void:
		var n := path.size()
		var rings := []
		var normals := []
		var side := side0
		_set_tint(path[0])
		for i in n:
			var t: Vector3
			if i == 0:
				t = (path[1] as Vector3) - (path[0] as Vector3)
			elif i == n - 1:
				t = (path[i] as Vector3) - (path[i - 1] as Vector3)
			else:
				t = (
					((path[i + 1] as Vector3) - (path[i] as Vector3)).normalized()
					+ ((path[i] as Vector3) - (path[i - 1] as Vector3)).normalized()
				)
			t = t.normalized()
			side = (side - t * side.dot(t))
			if side.length_squared() < 1e-8:
				side = t.cross(Vector3.UP if absf(t.y) < 0.9 else Vector3.RIGHT)
			side = side.normalized()
			var other := side.cross(t)
			var r: Vector2 = radii[i]
			var ring := []
			var nr := []
			for k in sides:
				var a := TAU * k / sides
				ring.append((path[i] as Vector3) + side * cos(a) * r.x + other * sin(a) * r.y)
				nr.append((side * cos(a) / r.x + other * sin(a) / r.y).normalized())
			rings.append(ring)
			normals.append(nr)
		for i in n - 1:
			for k in sides:
				var l := (k + 1) % sides
				tri(
					rings[i][k],
					rings[i][l],
					rings[i + 1][l],
					normals[i][k],
					normals[i][l],
					normals[i + 1][l],
					0.0
				)
				tri(
					rings[i][k],
					rings[i + 1][l],
					rings[i + 1][k],
					normals[i][k],
					normals[i + 1][l],
					normals[i + 1][k],
					0.0
				)
		var t0 := ((path[0] as Vector3) - (path[1] as Vector3)).normalized()
		var t1 := ((path[n - 1] as Vector3) - (path[n - 2] as Vector3)).normalized()
		for k in range(1, sides - 1):
			tri(rings[0][0], rings[0][k], rings[0][k + 1], t0, t0, t0, 0.0)
			tri(rings[n - 1][0], rings[n - 1][k], rings[n - 1][k + 1], t1, t1, t1, 0.0)

	## Round tube (straps, slings, barrels).
	func tube(path: Array, r: float, sides := 6) -> void:
		var radii := []
		for p: Vector3 in path:
			radii.append(Vector2(r, r))
		sweep(
			path,
			radii,
			sides,
			(
				Vector3.UP
				if absf(((path[1] as Vector3) - (path[0] as Vector3)).normalized().y) < 0.9
				else Vector3.RIGHT
			)
		)


class Kits:
	extends RefCounted
	var kits := {}

	func k(material: StringName) -> Kit:
		if not kits.has(material):
			kits[material] = Kit.new()
		return kits[material]

	func add_to(parent: Node3D, materials: Dictionary) -> void:
		for name: StringName in kits:
			var kit: Kit = kits[name]
			if kit.count > 0:
				ModelParts.add(parent, kit.commit(), materials[name], Vector3.ZERO)


static func build(faction: FactionDefinition) -> Node3D:
	var g := Node3D.new()
	var mats := _materials(faction)
	for i in SQUAD.size():
		var entry: Array = SQUAD[i]
		var s := Node3D.new()
		s.name = "Soldier%d" % i
		s.position = Vector3(entry[0], 0.0, entry[1])
		s.scale = Vector3.ONE * SOLDIER_SCALE
		s.rotation.y = (TerrainNoise.hash(i, 3, 1) - 0.5) * 0.6
		g.add_child(s)
		var rig := Node3D.new()
		rig.name = "Rig"
		s.add_child(rig)
		var kits := Kits.new()
		_soldier(kits, entry[2], entry[3])
		kits.add_to(rig, mats)
	return g


static func _paint(
	color: Color, mode: int, params: Dictionary, faction: FactionDefinition = null
) -> ShaderMaterial:
	var mat := ShaderMaterial.new()
	mat.shader = PAINT
	var camo := ModelParts.camo_colors(faction)
	mat.set_shader_parameter(&"color_a", color if mode == 1 else camo[0])
	mat.set_shader_parameter(&"color_b", camo[1])
	mat.set_shader_parameter(&"color_c", camo[2])
	mat.set_shader_parameter(&"pattern_mode", mode)
	var defaults := {
		"ground_ao_height": 0.03,
		"mud_height": 0.025,
		"streak_amount": 0.0,
		"wear_amount": 0.0,
		"dust_amount": 0.25,
		"fade": 0.12,
		"metallic_value": 0.0,
		"base_roughness": 0.9,
	}
	defaults.merge(params, true)
	for key: String in defaults:
		mat.set_shader_parameter(StringName(key), defaults[key])
	return mat


static func _materials(faction: FactionDefinition) -> Dictionary:
	var vest := faction.vest_color if faction != null else ModelParts.NEUTRAL_VEST
	return {
		&"uniform": _paint(Color.WHITE, 0, {"pattern_scale": 80.0, "tint_amount": 0.06}, faction),
		&"vest": _paint(vest, 1, {"tint_amount": 0.25}),
		&"dark":
		_paint(
			Color("#2a2b25"), 1, {"base_roughness": 0.8, "wear_amount": 0.3, "tint_amount": 0.2}
		),
		&"gun":
		_paint(
			Color("#1c1d1a"),
			1,
			{
				"base_roughness": 0.5,
				"metallic_value": 0.4,
				"wear_amount": 0.7,
				"dust_amount": 0.1,
				"fade": 0.0
			}
		),
		&"skin": ModelParts.std_material(Color("#a87c5f"), 0.65, 0.0),
		&"glass": ModelParts.std_material(Color("#14191c"), 0.08, 0.5),
		&"accent": ModelParts.accent(faction),
	}

static func _soldier(k: Kits, kneel: bool, kit: int) -> void:
	var up := Vector3(0.04, HIP_KNEEL - HIP, 0.0) if kneel else Vector3.ZERO
	if kneel:
		_leg(
			k,
			Vector3(0.0, HIP_KNEEL, -0.1),
			Vector3(0.44, 0.55, -0.12),
			Vector3(0.42, 0.12, -0.13),
			Basis.IDENTITY,
			false
		)
		_leg(
			k,
			Vector3(0.0, HIP_KNEEL, 0.1),
			Vector3(0.06, 0.09, 0.12),
			Vector3(-0.36, 0.17, 0.13),
			Basis(Vector3.BACK, -0.75),
			true
		)
	else:
		for side: float in [-1.0, 1.0]:
			var stride := -side * 0.06
			_leg(
				k,
				Vector3(0.0, HIP, side * 0.1),
				Vector3(stride + 0.02, 0.5, side * 0.11),
				Vector3(stride * 1.6, 0.12, side * 0.12),
				Basis.IDENTITY,
				false
			)
	_torso(k, up)
	_head(k, up)
	_gear(k, up)
	var hands: Array[Vector3]
	match kit:
		RPG:
			hands = _rpg7(k, Transform3D(Basis(Vector3.BACK, 0.04), up + Vector3(0.02, 1.49, 0.16)))
		MG:
			hands = _mg(k, up)
		_:
			hands = _rifle(k, up, -0.25 if kneel else -0.45)
	_arms(k, up, hands)


## Thigh and shin as one smooth sweep with knee and calf shaping, knee pad, cargo pocket, boot.
static func _leg(
	k: Kits, hip: Vector3, knee: Vector3, ankle: Vector3, foot_rot: Basis, knee_down: bool
) -> void:
	var uni := k.k(&"uniform")
	var path := [
		hip + (hip - knee).normalized() * 0.04,
		hip,
		hip.lerp(knee, 0.35),
		hip.lerp(knee, 0.75),
		knee,
		knee.lerp(ankle, 0.3),
		knee.lerp(ankle, 0.75),
		ankle + (knee - ankle).normalized() * 0.04
	]
	var radii := [
		Vector2(0.09, 0.09),
		Vector2(0.095, 0.095),
		Vector2(0.088, 0.09),
		Vector2(0.072, 0.074),
		Vector2(0.066, 0.068),
		Vector2(0.066, 0.07),
		Vector2(0.052, 0.054),
		Vector2(0.05, 0.05)
	]
	uni.sweep(path, radii, 12)
	var thigh := (knee - hip).normalized()
	var shin := (ankle - knee).normalized()
	var side := Vector3(0, 0, signf(hip.z))
	# Cargo pocket on the outer thigh.
	var pocket_basis := Basis(thigh.cross(side).normalized(), -thigh, side).orthonormalized()
	k.k(&"uniform").pillow(
		hip.lerp(knee, 0.5) + side * 0.075, Vector3(0.11, 0.15, 0.04), pocket_basis, 0.35, 6, 10
	)
	# Knee pad: forward of the knee, between thigh and shin.
	var front := (
		(-thigh + shin).normalized() if not knee_down else Vector3(0.3, -1.0, 0.0).normalized()
	)
	var pad_dir := (thigh.cross(side)).normalized()
	if pad_dir.dot(Vector3(1, 0, 0)) < 0.0:
		pad_dir = -pad_dir
	var pad_c := knee + pad_dir * 0.055 + shin * 0.02
	if knee_down:
		pad_c = knee + Vector3(0.03, -0.045, 0.0)
	k.k(&"dark").pillow(
		pad_c,
		Vector3(0.05, 0.12, 0.1),
		Basis(Vector3.BACK, atan2(-shin.x, -shin.y) if not knee_down else 1.3),
		0.45,
		6,
		10
	)
	# Boot: shaft, rounded upper, chamfered sole.
	var dark := k.k(&"dark")
	dark.sweep(
		[ankle + (knee - ankle).normalized() * 0.15, ankle + (knee - ankle).normalized() * 0.02],
		[Vector2(0.058, 0.06), Vector2(0.06, 0.062)],
		10
	)
	var foot := foot_rot
	var sole_c := ankle + foot * Vector3(0.05, -0.105, 0.0)
	dark.pillow(
		ankle + foot * Vector3(0.05, -0.05, 0.0), Vector3(0.28, 0.1, 0.11), foot, 0.45, 6, 12
	)
	dark.cbox(sole_c, Vector3(0.29, 0.03, 0.115), 0.01, foot)


static func _torso(k: Kits, up: Vector3) -> void:
	var uni := k.k(&"uniform")
	var path := []
	var radii := []
	for row: Array in [
		[0.84, 0.0, 0.15, 0.1],
		[0.92, 0.0, 0.17, 0.115],
		[1.02, 0.0, 0.155, 0.105],
		[1.14, 0.01, 0.165, 0.11],
		[1.27, 0.02, 0.19, 0.125],
		[1.37, 0.02, 0.2, 0.115],
		[1.43, 0.02, 0.17, 0.09],
		[1.47, 0.02, 0.08, 0.06],
	]:
		path.append(up + Vector3(row[1], row[0], 0.0))
		radii.append(Vector2(row[2], row[3]))
	uni.sweep(path, radii, 16)
	for side: float in [-1.0, 1.0]:
		uni.pillow(
			up + Vector3(0.0, 1.405, side * 0.2),
			Vector3(0.14, 0.13, 0.13),
			Basis.IDENTITY,
			0.9,
			6,
			10
		)
	# Collar of the shirt around the neck.
	uni.sweep(
		[up + Vector3(0.015, 1.45, 0.0), up + Vector3(0.02, 1.5, 0.0)],
		[Vector2(0.075, 0.068), Vector2(0.07, 0.064)],
		12
	)


static func _head(k: Kits, up: Vector3) -> void:
	var skin := k.k(&"skin")
	skin.sweep(
		[up + Vector3(0.015, 1.47, 0.0), up + Vector3(0.03, 1.57, 0.0)],
		[Vector2(0.052, 0.05), Vector2(0.05, 0.048)],
		10
	)
	var head := up + Vector3(0.035, 1.625, 0.0)
	skin.pillow(head, Vector3(0.19, 0.23, 0.16), Basis.IDENTITY, 0.9, 8, 14)
	skin.pillow(
		head + Vector3(0.05, -0.075, 0.0), Vector3(0.1, 0.08, 0.12), Basis.IDENTITY, 0.8, 5, 10
	)  # jaw
	skin.cbox(
		head + Vector3(0.098, -0.01, 0.0),
		Vector3(0.03, 0.05, 0.026),
		0.008,
		Basis(Vector3.BACK, 0.25)
	)  # nose
	for side: float in [-1.0, 1.0]:
		skin.pillow(
			head + Vector3(0.0, -0.005, side * 0.08),
			Vector3(0.04, 0.06, 0.02),
			Basis.IDENTITY,
			0.8,
			4,
			8
		)  # ears
	# Ballistic glasses.
	k.k(&"glass").cbox(
		head + Vector3(0.085, 0.022, 0.0),
		Vector3(0.03, 0.035, 0.15),
		0.008,
		Basis(Vector3.BACK, 0.1)
	)
	k.k(&"dark").tube(
		[head + Vector3(0.07, 0.024, -0.078), head + Vector3(-0.02, 0.03, -0.085)], 0.005, 4
	)
	k.k(&"dark").tube(
		[head + Vector3(0.07, 0.024, 0.078), head + Vector3(-0.02, 0.03, 0.085)], 0.005, 4
	)
	# wz. 2005 helmet with cover (camo), band, NVG shroud and chin strap.

## Upper arm, forearm (smooth, bent at the elbow), glove and the faction patch.
static func _arms(k: Kits, up: Vector3, hands: Array[Vector3]) -> void:
	var uni := k.k(&"uniform")
	for i in 2:
		var side := -1.0 if i == 0 else 1.0
		var shoulder := up + Vector3(0.0, 1.41, side * 0.21)
		var hand: Vector3 = hands[i]
		var elbow := (shoulder + hand) / 2.0 + Vector3(-0.08, -0.12, side * 0.07)
		var wrist := hand + (elbow - hand).normalized() * 0.06
		uni.sweep(
			[shoulder, shoulder.lerp(elbow, 0.5), elbow, elbow.lerp(wrist, 0.4), wrist],
			[
				Vector2(0.064, 0.064),
				Vector2(0.056, 0.058),
				Vector2(0.048, 0.05),
				Vector2(0.045, 0.046),
				Vector2(0.036, 0.036)
			],
			10
		)
		var fore := (hand - elbow).normalized()
		var glove_basis := (
			Basis(fore, fore.cross(Vector3.BACK * side).normalized(), Vector3.BACK)
			. orthonormalized()
		)
		k.k(&"dark").pillow(hand, Vector3(0.1, 0.05, 0.085), glove_basis, 0.6, 6, 10)
		var patch := shoulder.lerp(elbow, 0.38) + Vector3(0.0, 0.0, side * 0.058)
		k.k(&"accent").cbox(patch, Vector3(0.06, 0.04, 0.006), 0.002)


# --- Weapons ------------------------------------------------------------------------------------


static func _on(t: Transform3D, pos: Vector3, roll := 0.0) -> Array:
	return [t * pos, t.basis * Basis(Vector3.BACK, roll)]


static func _part(
	kit: Kit, t: Transform3D, pos: Vector3, size: Vector3, roll := 0.0, b := 0.006
) -> void:
	var o: Array = _on(t, pos, roll)
	kit.cbox(o[0], size, b, o[1])
